function exportReport() { const rows = document.querySelectorAll("#historyBody tr"); if (!rows.length || rows[0].querySelector(".empty")) { alert("No usage history available."); return; } let report = "AIHUBX - AI COST OPTIMIZER\n"; report += "================================\n\n"; report += "Dashboard Summary\n"; report += "Total Records: " + document.getElementById("totalRecords").innerText + "\n"; report += "Total API Calls: " + document.getElementById("dashboardApiCalls").innerText + "\n"; report += "Total Tokens: " + document.getElementById("dashboardTokens").innerText + "\n"; report += "Total AI Cost: " + document.getElementById("dashboardCost").innerText + "\n"; report += "Total Savings: " + document.getElementById("dashboardSavings").innerText + "\n\n"; report += "Usage History\n"; report += "================================\n"; rows.forEach(row => { const cells = row.querySelectorAll("td"); if (cells.length >= 7) { report += "Date: " + cells[0].innerText + "\n" + "Agent: " + cells[1].innerText + "\n" + "Model: " + cells[2].innerText + "\n" + "API Calls: " + cells[3].innerText + "\n" + "Tokens: " + cells[4].innerText + "\n" + "Cost: " + cells[5].innerText + "\n" + "Saving: " + cells[6].innerText + "\n\n"; } }); report += "================================\n"; report += "AIHUBX © 2026\n"; const blob = new Blob( [report], { type: "text/plain;charset=utf-8" } ); const url = URL.createObjectURL(blob); const link = document.createElement("a"); link.href = url; link.download = "AIHUBX-Usage-Report.txt"; document.body.appendChild(link); link.click(); link.remove(); URL.revokeObjectURL(url); }
