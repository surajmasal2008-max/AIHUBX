async function loadModelComparison() { const tokens = Number( document .getElementById("tokensInput") .value ); const container = document.getElementById( "modelComparison" ); if (!tokens || tokens <= 0) { return; } try { const response = await fetch( API_URL + "/models?tokens=" + encodeURIComponent(tokens) ); if (!response.ok) { throw new Error( "Model comparison failed" ); } const data = await response.json(); container.innerHTML = ""; data.models.forEach( function(item) { const row = document.createElement( "div" ); row.className = "model-row " + ( item.recommended ? "best-model" : "" ); row.innerHTML = `
${item.model} ${ item.recommended ? ` RECOMMENDED ` : "" }
₹${Number( item.cost ).toFixed(4)}
`; container.appendChild(row); } ); } catch (error) { console.error( "Model comparison error:", error ); container.innerHTML = `
Model comparison unavailable.
`; } }
